Готовый скрипт захвата UTM, Click ID, AI-поисковиков и органики для Bitrix24

Скрипт автономно определяет источник захода селлера (Google Ads, Meta Ads, SEO Google/Yandex, ChatGPT/Claude, Каталоги), сохраняет First-Touch и Last-Touch параметры в Cookies/LocalStorage и автоматически вставляет их в скрытые поля всех форм сайта перед отправкой в Битрикс24. 

/**
 * FFB Full Attribution Tracker v2.0
 * Автономный захват UTM, Click ID (gclid, fbclid), AI-поисковиков и рефереров.
 */
(function() {
  const COOKIE_EXPIRY_DAYS = 90;

  function getQueryParam(param) {
    const urlParams = new URLSearchParams(window.location.search);
    return urlParams.get(param) || '';
  }

  function detectChannel() {
    const utmSource = getQueryParam('utm_source');
    const gclid = getQueryParam('gclid');
    const fbclid = getQueryParam('fbclid');
    const referrer = document.referrer.toLowerCase();

    if (gclid) return { source: 'google', medium: 'cpc', campaign: getQueryParam('utm_campaign') || 'google_ads', click_id: gclid };
    if (fbclid) return { source: 'facebook', medium: 'cpc', campaign: getQueryParam('utm_campaign') || 'meta_ads', click_id: fbclid };
    if (utmSource) return { source: utmSource, medium: getQueryParam('utm_medium') || 'cpc', campaign: getQueryParam('utm_campaign') || 'campaign', click_id: '' };

    if (referrer.includes('chatgpt.com') || referrer.includes('perplexity.ai') || referrer.includes('claude.ai')) {
      return { source: 'ai_assistant', medium: 'organic_ai', campaign: 'ai_search', click_id: '' };
    }
    if (referrer.includes('google.')) return { source: 'google', medium: 'organic', campaign: 'seo', click_id: '' };
    if (referrer.includes('yandex.')) return { source: 'yandex', medium: 'organic', campaign: 'seo', click_id: '' };
    if (referrer) {
      try {
        const refDomain = new URL(referrer).hostname;
        return { source: refDomain, medium: 'referral', campaign: 'catalog', click_id: '' };
      } catch(e) {}
    }
    return { source: 'direct', medium: 'none', campaign: 'direct', click_id: '' };
  }

  const attribution = detectChannel();
  localStorage.setItem('ffb_last_attribution', JSON.stringify(attribution));
  if (!localStorage.getItem('ffb_first_attribution')) {
    localStorage.setItem('ffb_first_attribution', JSON.stringify(attribution));
  }

  // Инъекция во все формы
  document.addEventListener('submit', function(e) {
    const form = e.target;
    if (form.tagName === 'FORM') {
      const data = JSON.parse(localStorage.getItem('ffb_last_attribution') || '{}');
      for (const [key, val] of Object.entries(data)) {
        let input = form.querySelector('input[name="' + key + '"]');
        if (!input) {
          input = document.createElement('input');
          input.type = 'hidden';
          input.name = key;
          form.appendChild(input);
        }
        input.value = val;
      }
    }
  }, true);
})();
